% Create a custom coversheet
\newcommand{\makecoversheet}{
  \begin{titlepage}
    \centering
    \vspace*{2cm}
    {\Huge\bfseries $title$\par}
    \vspace{1.5cm}
    {\Large\itshape $author$\par}
    \vspace{1cm}
    {\large \today\par}
    \vspace{2cm}
    
    \begin{mdframed}[backgroundcolor=coverboxcolor, linewidth=1pt, roundcorner=10pt]
      \begin{center}
        \Large\textbf{Document Information}
      \end{center}
      \vspace{0.5cm}
      \begin{description}
        $if(subtitle)$
        \item[Subtitle:] $subtitle$
        $endif$
        $if(version)$
        \item[Version:] $version$
        $endif$
        $if(status)$
        \item[Status:] $status$
        $endif$
        $if(date)$
        \item[Date:] $date$
        $endif$
        $if(keywords)$
        \item[Keywords:] $keywords$
        $endif$
      \end{description}
    \end{mdframed}
    
    \vfill
    $if(logo)$
    \includegraphics[width=0.4\textwidth]{$logo$}
    $endif$
  \end{titlepage}
}
